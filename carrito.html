import { db } from "./firebase-config.js"; import { collection, addDoc, serverTimestamp } from "https://www.gstatic.com/firebasejs/10.13.2/firebase-firestore.js"; // ================= CONFIGURACIÓN ================= const CART_KEY = "vidaspa_carrito"; const WHATSAPP = "593959509052"; // ✅ NÚMERO CORRECTO (sin 0, sin +) const $ = id => document.getElementById(id); const items = $("cart-items"); const empty = $("cart-empty"); const totalHTML = $("cart-total"); const sumCount = $("sum-count"); const form = $("checkout-form"); const mensaje = $("checkout-message"); const confirmar = $("checkout-btn"); const cancelar = $("cancel-order-btn"); const metodoPagoSelect = $("metodo-pago"); const paymentPanel = $("payment-panel"); // ================= CARRITO ================= const getCart = () => JSON.parse(localStorage.getItem(CART_KEY)) || []; function saveCart(cart) { localStorage.setItem(CART_KEY, JSON.stringify(cart)); render(); } const money = n => `$${Number(n || 0).toFixed(2)}`; function fechaBonita(fecha) { if (!fecha) return "Por seleccionar"; return new Date(`${fecha}T12:00:00`) .toLocaleDateString("es-EC", { day: "2-digit", month: "long", year: "numeric" }); } function hoy() { return new Date().toISOString().split("T")[0]; } // ================= ACTUALIZAR ITEM ================= function actualizar(index, campo, valor) { const cart = getCart(); const item = cart[index]; if (!item) return; if (campo === "fecha") item.fecha = valor; if (campo === "hora") item.hora = valor; saveCart(cart); } // ================= MOSTRAR CARRITO ================= function render() { const cart = getCart(); items.innerHTML = ""; empty.classList.toggle("hidden", cart.length > 0); confirmar.disabled = cart.length === 0; cancelar.disabled = cart.length === 0; cart.forEach((item, index) => { items.innerHTML += `
${item.categoria || "Servicio"}

${item.nombre}

${item.descripcion || ""} · ${item.duracion || ""}

📅 ${fechaBonita(item.fecha)} · 🕒 ${item.hora || "Por seleccionar"}
Precio del servicio ${money(item.precio)}
`; }); eventos(); totales(cart); } // ================= EVENTOS ================= function eventos() { document.querySelectorAll(".cart-date").forEach(input => input.addEventListener("change", () => actualizar(+input.dataset.index, "fecha", input.value) ) ); document.querySelectorAll(".cart-time").forEach(select => select.addEventListener("change", () => actualizar(+select.dataset.index, "hora", select.value) ) ); document.querySelectorAll(".cart-remove").forEach(btn => btn.addEventListener("click", () => { const cart = getCart(); cart.splice(+btn.dataset.index, 1); saveCart(cart); }) ); } // ================= TOTALES ================= function totales(cart) { const total = cart.reduce((s, item) => s + Number(item.precio || 0), 0); sumCount.textContent = cart.length; totalHTML.textContent = money(total); } // ================= PANEL DE PAGO ================= function renderPaymentPanel(metodo) { if (!metodo) { paymentPanel.classList.add("hidden"); paymentPanel.innerHTML = ""; return; } paymentPanel.classList.remove("hidden"); if (metodo === "efectivo") { paymentPanel.innerHTML = `

💵 Pago en efectivo

Paga directamente en recepción el día de tu cita.

`; return; } if (metodo === "transferencia") { paymentPanel.innerHTML = `

🏦 Transferencia bancaria

BancoBanco Pichincha
CuentaN.º 2200XXXXXX
A nombre deVIDA SPA S.A.S.
Envía el comprobante por WhatsApp al confirmar. `; return; } if (metodo === "tarjeta") { paymentPanel.innerHTML = `

💳 Pago con tarjeta

Simulación de pago con tarjeta.

Formulario de demostración. `; return; } } metodoPagoSelect.addEventListener("change", () => { renderPaymentPanel(metodoPagoSelect.value); }); // ================= CANCELAR ================= cancelar.addEventListener("click", () => { if (!getCart().length) return; if (confirm("¿Deseas vaciar el carrito?")) { localStorage.removeItem(CART_KEY); location.href = "index.html#servicios"; } }); // ================= CONFIRMAR ================= confirmar.addEventListener("click", async () => { mensaje.textContent = ""; mensaje.classList.remove("error"); const cart = getCart(); if (!cart.length) { mensaje.textContent = "El carrito está vacío."; mensaje.classList.add("error"); return; } if (cart.some(item => !item.fecha || !item.hora)) { mensaje.textContent = "Selecciona fecha y horario para todos los servicios."; mensaje.classList.add("error"); return; } if (!form.checkValidity()) { mensaje.textContent = "Completa correctamente tus datos."; mensaje.classList.add("error"); form.reportValidity(); return; } const nombre = $("nombre").value.trim(); const correo = $("correo").value.trim(); const telefono = $("telefono").value.trim(); const metodoPago = metodoPagoSelect.value; const total = cart.reduce((s, item) => s + Number(item.precio || 0), 0); confirmar.disabled = true; confirmar.textContent = "Guardando..."; try { await addDoc(collection(db, "citas"), { nombre, correo, telefono, metodoPago, interes: cart.map(i => i.nombre).join(", "), servicios: cart, total, estado: "pendiente", origen: "carrito", creadoEn: serverTimestamp() }); const detalle = cart.map((item, i) => ` ${i + 1}. ${item.nombre} 📅 Fecha: ${fechaBonita(item.fecha)} 🕐 Hora: ${item.hora} 💰 Precio: ${money(item.precio)} `).join("\n"); const texto = `Hola VIDA SPA. Soy ${nombre}. Quiero confirmar mi cita: ${detalle} 💳 Método de pago: ${metodoPago} 💰 TOTAL: ${money(total)} 📞 Teléfono: ${telefono} ✉️ Correo: ${correo}`; localStorage.removeItem(CART_KEY); // ✅ REDIRIGIR A WHATSAPP CON EL NÚMERO CORRECTO const urlWhatsApp = `https://wa.me/${WHATSAPP}?text=${encodeURIComponent(texto)}`; console.log("🔗 Abriendo WhatsApp:", urlWhatsApp); window.location.href = urlWhatsApp; } catch (error) { console.error("❌ Error:", error); mensaje.textContent = "❌ No se pudo guardar la solicitud."; mensaje.classList.add("error"); confirmar.disabled = false; confirmar.textContent = "Confirmar cita por WhatsApp"; } }); // ================= INICIO ================= render(); console.log("✅ Carrito de VIDA SPA cargado"); console.log(`📱 Número de WhatsApp: +${WHATSAPP}`);
